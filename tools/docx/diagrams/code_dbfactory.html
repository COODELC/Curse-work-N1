 2 // Фабрика контекста: SQL Server или локальный файл SQLite
 3 public sealed class DbFactory
 4 {
 5     private readonly string _connectionString;
 6 
 7     public bool IsLocalFileDatabase =>
 8         _connectionString.StartsWith("Data Source=",
 9             StringComparison.OrdinalIgnoreCase);
10 
11     public WorkshopDbContext Create()
12     {
13         var builder = new DbContextOptionsBuilder<WorkshopDbContext>();
14 
15         if (IsLocalFileDatabase)
16         {
17             var path = Path.Combine(AppContext.BaseDirectory, "autoshop.db");
18             builder.UseSqlite($"Data Source={path}");
19         }
20         else
21         {
22             builder.UseSqlServer(_connectionString);
23         }
24 
25         return new WorkshopDbContext(builder.Options);
26     }
27 }
